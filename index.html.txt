  <!-- JavaScript for Interactive Dropdown Stock Update -->
  <script>
    function updateStock(selectElement, targetId) {
      const selectedValue = selectElement.value;
      const targetBadge = document.getElementById(targetId);
      
      targetBadge.innerText = selectedValue;
      
      // Highlight critical/low stock
      if (selectedValue.includes("0 Units") || selectedValue.includes("1 Unit")) {
        targetBadge.className = "badge bg-warning text-dark fs-6";
      } else {
        targetBadge.className = "badge bg-danger fs-6";
      }
    }
  </script>
